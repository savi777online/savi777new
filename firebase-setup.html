2 Create Firestore Database

3 Add Sample Market Data

Run the setup button above to create sample markets

4 Add Admin User

Register a user through the app, then manually set their role to "admin" in Firestore

🗂️ Required Firestore Indexes

If you see errors about missing indexes, create these composite indexes:

Index 1: bids collection

{ "indexes": [ { "collectionGroup": "bids", "queryScope": "COLLECTION", "fields": [ { "fieldPath": "userId", "order": "ASCENDING" }, { "fieldPath": "createdAt", "order": "DESCENDING" } ] } ] }

Index 2: transactions collection

{ "indexes": [ { "collectionGroup": "transactions", "queryScope": "COLLECTION", "fields": [ { "fieldPath": "userId", "order": "ASCENDING" }, { "fieldPath": "date", "order": "DESCENDING" } ] } ] }

Index 3: withdrawRequests collection

{ "indexes": [ { "collectionGroup": "withdrawRequests", "queryScope": "COLLECTION", "fields": [ { "fieldPath": "userId", "order": "ASCENDING" }, { "fieldPath": "date", "order": "DESCENDING" } ] } ] }
⚠️ How to add indexes:
1. Go to Firebase Console → Firestore Database → Indexes tab
2. Click "Add Index" or "Import JSON"
3. Paste the JSON above
4. Wait for indexes to build (may take a few minutes)

🔒 Firestore Security Rules

Copy these rules to Firebase Console → Firestore Database → Rules

rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // Users collection match /users/{userId} { allow read, write: if request.auth != null && request.auth.uid == userId; allow create: if request.auth != null; } // Markets collection (public read) match /markets/{market} { allow read: if true; allow write: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } // Bids collection match /bids/{bid} { allow read: if request.auth != null && (resource.data.userId == request.auth.uid || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'); allow create: if request.auth != null; allow update: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } // Transactions collection match /transactions/{transaction} { allow read: if request.auth != null && (resource.data.userId == request.auth.uid || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'); allow create: if request.auth != null; allow update: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } // Withdraw requests collection match /withdrawRequests/{request} { allow read: if request.auth != null && (resource.data.userId == request.auth.uid || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'); allow create: if request.auth != null; allow update: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } // Chats collection match /chats/{chat} { allow read, write: if request.auth != null; } match /chats/{chat}/messages/{message} { allow read, write: if request.auth != null; } // Settings collection (admin only) match /settings/{setting} { allow read: if true; allow write: if request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'; } } }
